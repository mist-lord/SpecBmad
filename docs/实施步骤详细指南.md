```typescript
import { Command } from 'commander';
import { Logger } from '../utils/logger';
import { ConfigManager } from '../utils/config';
import { ProjectState } from '../types/project';

export abstract class BaseCommand {
  protected logger: Logger;
  protected config: ConfigManager;

  constructor() {
    this.logger = new Logger(this.constructor.name);
    this.config = new ConfigManager();
  }

  abstract execute(options: any): Promise<void>;

  protected async validateProject(): Promise<ProjectState> {
    const projectState = await this.config.getProjectState();
    if (!projectState) {
      throw new Error('Not in a speckit-bmad project. Run "speckit-bmad init" first.');
    }
    return projectState;
  }

  protected createCommand(name: string, description: string): Command {
    const command = new Command(name);
    command
      .description(description)
      .action(async (options) => {
        try {
          await this.execute(options);
        } catch (error) {
          this.logger.error(`Command ${name} failed:`, error);
          process.exit(1);
        }
      });
    return command;
  }
}
```

```typescript
import * as fs from 'fs/promises';
import * as path from 'path';
import { ProjectState, AgentConfig, LLMConfig } from '../types/project';

export class ConfigManager {
  private static readonly CONFIG_DIR = '.bmad-core';
  private static readonly STATE_FILE = 'state.json';
  private static readonly AGENTS_DIR = 'agents';

  async getProjectState(): Promise<ProjectState | null> {
    try {
      const statePath = path.join(process.cwd(), ConfigManager.CONFIG_DIR, ConfigManager.STATE_FILE);
      const stateContent = await fs.readFile(statePath, 'utf-8');
      return JSON.parse(stateContent);
    } catch (error) {
      return null;
    }
  }

  async updateProjectState(updates: Partial<ProjectState>): Promise<void> {
    const currentState = await this.getProjectState() || this.getDefaultProjectState();
    const newState = { ...currentState, ...updates, updatedAt: new Date().toISOString() };
    
    const statePath = path.join(process.cwd(), ConfigManager.CONFIG_DIR, ConfigManager.STATE_FILE);
    await fs.writeFile(statePath, JSON.stringify(newState, null, 2));
  }

  async getAgentConfig(agentName: string): Promise<AgentConfig | null> {
    try {
      const agentPath = path.join(process.cwd(), ConfigManager.CONFIG_DIR, ConfigManager.AGENTS_DIR, `${agentName}.md`);
      const agentContent = await fs.readFile(agentPath, 'utf-8');
      return this.parseAgentConfig(agentContent);
    } catch (error) {
      return null;
    }
  }

  async getLLMConfig(): Promise<LLMConfig> {
    const state = await this.getProjectState();
    return state?.llmConfig || this.getDefaultLLMConfig();
  }

  private getDefaultProjectState(): ProjectState {
    return {
      projectName: path.basename(process.cwd()),
      version: '1.0.0',
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      agents: [],
      llmConfig: this.getDefaultLLMConfig(),
      workflow: {
        currentStep: 'init',
        completedSteps: [],
        artifacts: {}
      }
    };
  }

  private getDefaultLLMConfig(): LLMConfig {
    return {
      defaultProvider: 'openai',
      providers: {
        openai: {
          apiKey: process.env.OPENAI_API_KEY || '',
          model: 'gpt-4',
          baseURL: 'https://api.openai.com/v1'
        },
        anthropic: {
          apiKey: process.env.ANTHROPIC_API_KEY || '',
          model: 'claude-3-sonnet-20240229',
          baseURL: 'https://api.anthropic.com'
        }
      }
    };
  }

      throw new Error('Invalid agent config format');

    // 这里需要YAML解析器，暂时用JSON格式

    return {
      name: config.name,
      role: config.role,
      capabilities: config.capabilities || [],
      model: config.model || 'gpt-4',
      systemPrompt: content.substring(yamlMatch[0].length).trim()
    };
  }
```

```typescript
export enum LogLevel {
  DEBUG = 0,
  INFO = 1,
  WARN = 2,
  ERROR = 3
}

export class Logger {
  private static globalLevel: LogLevel = LogLevel.INFO;
  private context: string;

  constructor(context: string = 'App') {
    this.context = context;
  }

  static setLevel(level: string): void {
    switch (level.toLowerCase()) {
      case 'debug': Logger.globalLevel = LogLevel.DEBUG; break;
      case 'info': Logger.globalLevel = LogLevel.INFO; break;
      case 'warn': Logger.globalLevel = LogLevel.WARN; break;
      case 'error': Logger.globalLevel = LogLevel.ERROR; break;
    }
  }

  debug(message: string, ...args: any[]): void {
    this.log(LogLevel.DEBUG, message, ...args);
  }

  info(message: string, ...args: any[]): void {
    this.log(LogLevel.INFO, message, ...args);
  }

  warn(message: string, ...args: any[]): void {
    this.log(LogLevel.WARN, message, ...args);
  }

  error(message: string, ...args: any[]): void {
    this.log(LogLevel.ERROR, message, ...args);
  }

  private log(level: LogLevel, message: string, ...args: any[]): void {
    if (level < Logger.globalLevel) return;

    const timestamp = new Date().toISOString();
    const levelName = LogLevel[level];
    const prefix = `[${timestamp}] [${levelName}] [${this.context}]`;
    
    switch (level) {
      case LogLevel.DEBUG:
        console.debug(prefix, message, ...args);
        break;
      case LogLevel.INFO:
        console.info(prefix, message, ...args);
        break;
      case LogLevel.WARN:
        console.warn(prefix, message, ...args);
        break;
      case LogLevel.ERROR:
        console.error(prefix, message, ...args);
        break;
    }
  }
}
```

```typescript
import { Agent, AgentContext, AgentResult, AgentMemory } from '../../types/agent';
import { LLMClient } from '../../llm/base-client';
import { Logger } from '../../utils/logger';

export abstract class BaseAgent implements Agent {
  public readonly name: string;
  public readonly role: string;
  public readonly capabilities: string[];
  
  protected llmClient: LLMClient;
  protected logger: Logger;
  protected memory: AgentMemory;

  constructor(
    name: string,
    role: string,
    capabilities: string[],
    llmClient: LLMClient
  ) {
    this.name = name;
    this.role = role;
    this.capabilities = capabilities;
    this.llmClient = llmClient;
    this.logger = new Logger(`Agent:${name}`);
    this.memory = this.initializeMemory();
  }

  abstract execute(context: AgentContext): Promise<AgentResult>;

  validate(input: any): boolean {
    // 基础验证逻辑
    return input !== null && input !== undefined;
  }

  protected async generateResponse(prompt: string, context: AgentContext): Promise<string> {
    const systemPrompt = this.buildSystemPrompt(context);
    const fullPrompt = `${systemPrompt}\n\n${prompt}`;
    
    this.logger.debug('Generating response with prompt:', fullPrompt);
    
    try {
      const response = await this.llmClient.generate(fullPrompt, {
        maxTokens: 2000,
        temperature: 0.7
      });
      
      this.updateMemory('lastResponse', response);
      return response;
    } catch (error) {
      this.logger.error('Failed to generate response:', error);
      throw error;
    }
  }

  protected buildSystemPrompt(context: AgentContext): string {
    return `You are ${this.name}, a ${this.role} with the following capabilities: ${this.capabilities.join(', ')}.
    
Project Context:
- Project: ${context.projectState.projectName}
- Current Step: ${context.projectState.workflow.currentStep}
- Completed Steps: ${context.projectState.workflow.completedSteps.join(', ')}

Your task is to provide professional, accurate, and actionable output based on your role and capabilities.`;
  }

  protected updateMemory(key: string, value: any): void {
    this.memory.shortTerm[key] = value;
    this.memory.context.push(`${new Date().toISOString()}: ${key} updated`);
    
    // 保持上下文历史在合理范围内
    if (this.memory.context.length > 100) {
      this.memory.context = this.memory.context.slice(-50);
    }
  }

  private initializeMemory(): AgentMemory {
    return {
      shortTerm: {},
      longTerm: {},
      context: []
    };
  }
}
```

```typescript
import { BaseAgent } from './base/agent';
import { AgentContext, AgentResult } from '../types/agent';
import { LLMClient } from '../llm/base-client';
import * as fs from 'fs/promises';
import * as path from 'path';

export class AnalystAgent extends BaseAgent {
  constructor(llmClient: LLMClient) {
    super(
      'Analyst',
      'Business Analyst',
      ['requirement-analysis', 'user-story-creation', 'business-process-modeling'],
      llmClient
    );
  }

  async execute(context: AgentContext): Promise<AgentResult> {
    this.logger.info('Starting business analysis...');

    try {
      const requirements = await this.analyzeRequirements(context);
      const userStories = await this.generateUserStories(requirements, context);
      const specification = await this.createSpecification(requirements, userStories, context);

      // 保存规格文档
      const specPath = path.join(context.workingDirectory, '.specify', 'spec.md');
      await fs.writeFile(specPath, specification);

      return {
        success: true,
        output: {
          requirements,
          userStories,
          specification
        },
        artifacts: [specPath],
        nextSteps: ['Create technical architecture plan'],
        metadata: {
          requirementsCount: requirements.length,
          userStoriesCount: userStories.length
        }
      };
    } catch (error) {
      this.logger.error('Analysis failed:', error);
      return {
        success: false,
        output: null,
        artifacts: [],
        errors: [error.message]
      };
    }
  }

  private async analyzeRequirements(context: AgentContext): Promise<string[]> {
    const prompt = `Based on the project "${context.projectState.projectName}", analyze and extract the key business requirements. 
    
Input data: ${JSON.stringify(context.inputData, null, 2)}

Please provide a comprehensive list of functional and non-functional requirements.`;

    const response = await this.generateResponse(prompt, context);
    
    // 解析响应中的需求列表
    const requirements = response.split('\n')
      .filter(line => line.trim().startsWith('-') || line.trim().startsWith('*'))
      .map(line => line.trim().substring(1).trim());

    return requirements;
  }

  private async generateUserStories(requirements: string[], context: AgentContext): Promise<string[]> {
    const prompt = `Convert the following requirements into user stories using the format "As a [user], I want [goal] so that [benefit]":

Requirements:
${requirements.map((req, index) => `${index + 1}. ${req}`).join('\n')}

Generate comprehensive user stories that cover all requirements.`;

    const response = await this.generateResponse(prompt, context);
    
    const userStories = response.split('\n')
      .filter(line => line.trim().toLowerCase().startsWith('as a'))
      .map(line => line.trim());

    return userStories;
  }

  private async createSpecification(requirements: string[], userStories: string[], context: AgentContext): Promise<string> {
    const template = await this.loadTemplate('spec-template.md');
    
    const specification = template
      .replace('{{PROJECT_NAME}}', context.projectState.projectName)
      .replace('{{REQUIREMENTS}}', requirements.map((req, i) => `${i + 1}. ${req}`).join('\n'))
      .replace('{{USER_STORIES}}', userStories.map((story, i) => `${i + 1}. ${story}`).join('\n'))
      .replace('{{CREATED_DATE}}', new Date().toISOString().split('T')[0]);

    return specification;
  }

  private async loadTemplate(templateName: string): Promise<string> {
    const templatePath = path.join(process.cwd(), 'templates', templateName);
    return await fs.readFile(templatePath, 'utf-8');
  }
}
```

```typescript
import { ProjectState, WorkflowState } from '../types/project';
import { AgentFactory } from '../agents/factory';

import { ConfigManager } from '../utils/config';
import { Logger } from '../utils/logger';

export class Orchestrator {
  private logger: Logger;
  private config: ConfigManager;
  private agents: Map<string, Agent>;

  constructor() {
    this.logger = new Logger('Orchestrator');
    this.config = new ConfigManager();
    this.agents = new Map();
  }

  async executeWorkflow(workflowName: string, inputData: any): Promise<AgentResult[]> {
    this.logger.info(`Starting workflow: ${workflowName}`);

    const projectState = await this.config.getProjectState();
    if (!projectState) {
      throw new Error('Project not initialized');
    }

    const workflow = this.getWorkflowDefinition(workflowName);
    const results: AgentResult[] = [];

    for (const step of workflow.steps) {
      this.logger.info(`Executing step: ${step.agent}`);

      try {
        const agent = await this.getAgent(step.agent);
        const context = await this.buildAgentContext(projectState, inputData, results);
        
        const result = await agent.execute(context);
        results.push(result);

        if (!result.success) {
          this.logger.error(`Step ${step.agent} failed:`, result.errors);
          break;
        }

        // 更新项目状态
        await this.updateWorkflowState(step.agent, result);

      } catch (error) {
        this.logger.error(`Step ${step.agent} execution error:`, error);
        results.push({
          success: false,
          output: null,
          artifacts: [],
          errors: [error.message]
        });
        break;
      }
    }

    return results;
  }

  private getWorkflowDefinition(workflowName: string): WorkflowDefinition {
    const workflows: Record<string, WorkflowDefinition> = {
      'full-development': {
        name: 'Full Development Workflow',
        steps: [
          { agent: 'analyst', dependencies: [] },
          { agent: 'architect', dependencies: ['analyst'] },
          { agent: 'scrum-master', dependencies: ['architect'] },
          { agent: 'developer', dependencies: ['scrum-master'] },
          { agent: 'qa', dependencies: ['developer'] }
        ]
      },
      'planning-only': {
        name: 'Planning Only Workflow',
        steps: [
          { agent: 'analyst', dependencies: [] },
          { agent: 'architect', dependencies: ['analyst'] },
          { agent: 'scrum-master', dependencies: ['architect'] }
        ]
      }
    };

    const workflow = workflows[workflowName];
    if (!workflow) {
      throw new Error(`Unknown workflow: ${workflowName}`);
    }

    return workflow;
  }

  private async getAgent(agentType: string): Promise<Agent> {
    if (!this.agents.has(agentType)) {
      const llmConfig = await this.config.getLLMConfig();
      const llmClient = LLMClientFactory.createClient(llmConfig.defaultProvider, llmConfig);
      const agent = AgentFactory.createAgent(agentType, llmClient);
      this.agents.set(agentType, agent);
    }

    return this.agents.get(agentType)!;
  }

  private async buildAgentContext(
    projectState: ProjectState,
    inputData: any,
    previousResults: AgentResult[]
  ): Promise<AgentContext> {
    return {
      projectState,
      inputData,
      previousResults,
      templates: {}, // TODO: 加载模板
      workingDirectory: process.cwd()
    };
  }

  private async updateWorkflowState(stepName: string, result: AgentResult): Promise<void> {
    const updates: Partial<ProjectState> = {
      workflow: {
        currentStep: stepName,
        completedSteps: [], // TODO: 从当前状态获取并添加
        artifacts: {} // TODO: 更新artifacts
      }
    };

    await this.config.updateProjectState(updates);
  }
```

```typescript
import { BaseLLMClient, GenerateOptions, ChatOptions, ChatMessage } from './base-client';
import { Logger } from '../utils/logger';

export class OpenAIClient extends BaseLLMClient {
  private logger: Logger;

  constructor(apiKey: string, model: string = 'gpt-4', baseURL: string = 'https://api.openai.com/v1') {
    super('openai', model, apiKey, baseURL);
    this.logger = new Logger('OpenAIClient');
  }

  async generate(prompt: string, options?: GenerateOptions): Promise<string> {
    const validatedOptions = this.validateOptions(options);
    
    const messages: ChatMessage[] = [];
    
    if (validatedOptions.systemPrompt) {
      messages.push({ role: 'system', content: validatedOptions.systemPrompt });
    }
    
    messages.push({ role: 'user', content: prompt });

    return this.chat(messages, validatedOptions);
  }

  async chat(messages: ChatMessage[], options?: ChatOptions): Promise<string> {
    const validatedOptions = this.validateOptions(options);

    try {
      const response = await fetch(`${this.baseURL}/chat/completions`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          model: this.model,
          messages: messages,
          max_tokens: validatedOptions.maxTokens,
          temperature: validatedOptions.temperature,
          stop: validatedOptions.stopSequences?.length ? validatedOptions.stopSequences : undefined
        })
      });

      if (!response.ok) {
        const error = await response.json();
        throw new Error(`OpenAI API error: ${error.error?.message || response.statusText}`);
      }

      const data = await response.json();
      
      if (!data.choices || data.choices.length === 0) {
        throw new Error('No response from OpenAI API');
      }

      const content = data.choices[0].message?.content;
      if (!content) {
        throw new Error('Empty response from OpenAI API');
      }

      this.logger.debug(`Generated response (${data.usage?.total_tokens} tokens)`);
      return content;

    } catch (error) {
      this.logger.error('OpenAI API call failed:', error);
      throw error;
    }
  }

  async validateApiKey(): Promise<boolean> {
    try {
      const response = await fetch(`${this.baseURL}/models`, {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`
        }
      });
      return response.ok;
    } catch {
      return false;
    }
  }
}
```

```typescript
    for (const option of cmd.options) {
      content += `- \`${option.flag}\`: ${option.description}\n`;
    }
    
    content += '\n';
  }

  await fs.writeFile('docs/api/commands.md', content);
}

generateAPIDocs().catch(console.error);
```
